const canvas = document.getElementById('gameCanvas'); const ctx = canvas.getContext('2d'); const scoreElement = document.getElementById('score'); const ammoElement = document.getElementById('ammoDisplay'); const gameOverScreen = document.getElementById('gameOverScreen'); const finalScoreElement = document.getElementById('finalScore'); const btnRestart = document.getElementById('btnRestart'); canvas.width = window.innerWidth; canvas.height = window.innerHeight; let score, gameOver, currentAmmo, isReloading, player, projectiles, enemies; const maxAmmo = 15; // Aumentei um pouco a munição para o modo automático const reloadTime = 1500; const fireRate = 150; // Velocidade do tiro automático (milissegundos) let isMouseDown = false; let fireTimer = null; let mousePos = { x: 0, y: 0 }; const keys = {}; function init() { score = 0; gameOver = false; currentAmmo = maxAmmo; isReloading = false; projectiles = []; enemies = []; player = { x: canvas.width / 2, y: canvas.height / 2, radius: 15, color: '#0095DD', speed: 5 }; scoreElement.innerHTML = score; updateAmmoUI(); gameOverScreen.style.display = 'none'; animate(); } // Rastrear posição do mouse para o tiro automático saber para onde olhar window.addEventListener('mousemove', (e) => { mousePos.x = e.clientX; mousePos.y = e.clientY; }); // Lógica de Tiro Automático window.addEventListener('mousedown', () => { isMouseDown = true; startFiring(); }); window.addEventListener('mouseup', () => { isMouseDown = false; stopFiring(); }); function startFiring() { if (fireTimer) return; // Evita duplicar o intervalo shoot(); // Atira o primeiro logo no clique fireTimer = setInterval(shoot, fireRate); } function stopFiring() { clearInterval(fireTimer); fireTimer = null; } function shoot() { if (gameOver || isReloading || currentAmmo <= 0 || !isMouseDown) { if (currentAmmo <= 0 && !isReloading) startReload(); return; } const angle = Math.atan2(mousePos.y - player.y, mousePos.x - player.x); projectiles.push({ x: player.x, y: player.y, radius: 5, color: 'yellow', velocity: { x: Math.cos(angle) * 10, y: Math.sin(angle) * 10 } }); currentAmmo--; updateAmmoUI(); if (currentAmmo <= 0) { stopFiring(); startReload(); } } function startReload() { isReloading = true; updateAmmoUI(); setTimeout(() => { currentAmmo = maxAmmo; isReloading = false; updateAmmoUI(); // Se o jogador ainda estiver segurando o mouse após recarregar, volta a atirar if (isMouseDown) startFiring(); }, reloadTime); } function updateAmmoUI() { ammoElement.innerHTML = isReloading ? "RECARREGANDO..." : `Munição: ${currentAmmo} / ${maxAmmo}`; } // Restante dos controles e lógica de movimento window.addEventListener('keydown', (e) => keys[e.key.toLowerCase()] = true); window.addEventListener('keyup', (e) => keys[e.key.toLowerCase()] = false); // Spawn de inimigos setInterval(() => { if(gameOver) return; let x, y; if (Math.random() < 0.5) { x = Math.random() < 0.5 ? -30 : canvas.width + 30; y = Math.random() * canvas.height; } else { x = Math.random() * canvas.width; y = Math.random() < 0.5 ? -30 : canvas.height + 30; } const angle = Math.atan2(player.y - y, player.x - x); enemies.push({ x, y, radius: 20, color: '#ff4d4d', velocity: { x: Math.cos(angle) * 2.5, y: Math.sin(angle) * 2.5 } }); }, 800); // Zumbis aparecem um pouco mais rápido agora function animate() { if(gameOver) return; requestAnimationFrame(animate); ctx.clearRect(0, 0, canvas.width, canvas.height); if (keys['w'] && player.y > 0) player.y -= player.speed; if (keys['s'] && player.y < canvas.height) player.y += player.speed; if (keys['a'] && player.x > 0) player.x -= player.speed; if (keys['d'] && player.x < canvas.width) player.x += player.speed; ctx.beginPath(); ctx.arc(player.x, player.y, player.radius, 0, Math.PI * 2); ctx.fillStyle = player.color; ctx.fill(); projectiles.forEach((p, i) => { p.x += p.velocity.x; p.y += p.velocity.y; ctx.beginPath(); ctx.arc(p.x, p.y, p.radius, 0, Math.PI * 2); ctx.fillStyle = p.color; ctx.fill(); if (p.x < 0 || p.x > canvas.width || p.y < 0 || p.y > canvas.height) projectiles.splice(i, 1); }); enemies.forEach((enemy, index) => { const angle = Math.atan2(player.y - enemy.y, player.x - enemy.x); enemy.x += Math.cos(angle) * 2; enemy.y += Math.sin(angle) * 2; ctx.beginPath(); ctx.arc(enemy.x, enemy.y, enemy.radius, 0, Math.PI * 2); ctx.fillStyle = enemy.color; ctx.fill(); if (Math.hypot(player.x - enemy.x, player.y - enemy.y) < player.radius + enemy.radius) { gameOver = true; finalScoreElement.innerHTML = `Pontuação Final: ${score}`; gameOverScreen.style.display = 'flex'; } projectiles.forEach((p, pIndex) => { if (Math.hypot(p.x - enemy.x, p.y - enemy.y) < p.radius + enemy.radius) { enemies.splice(index, 1); projectiles.splice(pIndex, 1); score += 10; scoreElement.innerHTML = score; } }); }); } btnRestart.addEventListener('click', init); init();
